# 10.2 스킬 만들기 — 모델이 알아서 부르는 명령

---

## 이 챕터에서 배우는 것

- 10.1의 슬래시 명령은 **사용자가 부른다**. 스킬은 **모델이 알아서 부른다** — 두 invoke 경로
- 디렉토리가 단위인 스킬 형식 — 부속 스크립트 + 설정 + 마크다운이 한 곳에 산다
- `when_to_use`가 모델의 결정에 직접 들어가는 자리 — **좋은 트리거 문장의 힘**
- **SkillTool**을 9.3의 Tool Protocol로 구현 — 스킬 시스템이 도구 풀에 한 줄로 합류
- 1% 컨텍스트 예산 — 스킬 목록은 발견용이지 내용 전달용이 아니다
- 진짜 코드의 충격: **슬래시 명령과 스킬은 모델 입장에서 이미 같은 것**

---

## 10.1에서 빠진 것

10.1의 명령은 사용자가 `/today`를 **기억하고 쳐야** 동작한다. 문제:

- 명령이 10개가 넘으면 사용자가 전부 기억 못 한다
- "이 PR 한 번 봐줘"라고 자연어로 말했는데 — 그게 `/pr-review` 명령이 있다는 걸 사용자가 모른다
- 부속 스크립트가 필요한 복잡한 명령은 마크다운 한 파일에 다 담기 어색하다

스킬이 이 세 가지를 푼다: **모델이 자동 발견** + **디렉토리 형식** + **두 invoke 경로**.

## 구현 — 3단계

### 1단계: Skill 데이터 클래스 + `SKILL.md` 로더

10.1의 `MarkdownCommand`에서 두 가지가 달라진다: **디렉토리 형식** + **`when_to_use`** 필드.

```python
# src/mini_claude/skills/skill_md.py
from __future__ import annotations
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

from ..commands.markdown_loader import (
    parse_frontmatter,
    substitute_arguments,
    _parse_allowed_tools,
)


@dataclass
class Skill:
    """SKILL.md가 변환된 스킬.

    10.1의 MarkdownCommand와 같은 혈통이지만 세 가지가 다르다:
    - 디렉토리가 단위 (평탄 .md 거부)
    - when_to_use 필드 (모델이 자동 발견)
    - ${CLAUDE_SKILL_DIR} (부속 파일 참조)
    """
    name: str                                    # 디렉토리 이름
    description: str
    when_to_use: str | None = None               # ⭐ 모델 결정에 들어가는 자연어
    body: str = ""
    base_dir: Path | None = None                 # ${CLAUDE_SKILL_DIR}
    allowed_tools: list[str] = field(default_factory=list)
    user_invocable: bool = True
    disable_model_invocation: bool = False
    argument_names: list[str] = field(default_factory=list)

    def get_prompt(self, args: str = "", session_id: str = "") -> str:
        """호출 시점에 본문을 변환한다.

        10.1의 get_prompt()에 두 가지 추가:
        - ${CLAUDE_SKILL_DIR} → 스킬 디렉토리 절대 경로
        - base_dir 안내 헤더 (모델이 부속 파일 위치를 안다)
        """
        content = self.body

        # ① 인자 치환 (10.1 함수 재사용)
        content = substitute_arguments(content, args, self.argument_names)

        # ② ${CLAUDE_SKILL_DIR} — 부속 스크립트 절대 경로 참조
        if self.base_dir:
            content = content.replace(
                "${CLAUDE_SKILL_DIR}",
                str(self.base_dir.absolute()),
            )

        # ③ ${CLAUDE_SESSION_ID}
        content = content.replace("${CLAUDE_SESSION_ID}", session_id)

        # ④ 모델에게 base_dir 알려주기
        if self.base_dir:
            content = (
                f"Base directory for this skill: {self.base_dir}\n\n{content}"
            )

        return content


def load_skill(skill_dir: Path) -> Skill | None:
    """디렉토리 한 개 → Skill 한 개.

    디렉토리가 아니거나 SKILL.md가 없으면 None.
    """
    if not skill_dir.is_dir():
        return None  # ⭐ 평탄 .md 거부

    skill_md = skill_dir / "SKILL.md"  # ⭐ 고정된 이름
    if not skill_md.exists():
        return None

    text = skill_md.read_text(encoding="utf-8")
    frontmatter, body = parse_frontmatter(text)

    description = frontmatter.get("description")
    if not description:
        first_line = next(
            (line.strip() for line in body.split("\n") if line.strip()),
            "",
        )
        description = first_line[:80]

    return Skill(
        name=skill_dir.name,             # ⭐ 디렉토리 이름 = 스킬 이름
        description=description,
        when_to_use=frontmatter.get("when_to_use"),
        body=body,
        base_dir=skill_dir,
        allowed_tools=_parse_allowed_tools(frontmatter.get("allowed-tools", "")),
        user_invocable=frontmatter.get("user-invocable", True),
        disable_model_invocation=frontmatter.get("disable-model-invocation", False),
        argument_names=str(frontmatter.get("arguments", "")).split(),
    )


def discover_skills(cwd: Path, home: Path) -> list[Skill]:
    """10.1의 discover_commands()와 같은 walker.

    차이: .claude/skills/ 디렉토리 안의 *서브 디렉토리*를 스킬로 시도한다.
    """
    dirs: list[Path] = []

    current = cwd.resolve()
    home_resolved = home.resolve()
    while current != current.parent and current != home_resolved:
        skills_dir = current / ".claude" / "skills"
        if skills_dir.is_dir():
            dirs.append(skills_dir)
        if (current / ".git").exists():
            break
        current = current.parent

    user_skills = home_resolved / ".claude" / "skills"
    if user_skills.is_dir():
        dirs.append(user_skills)

    seen: dict[str, Skill] = {}
    for d in reversed(dirs):
        for entry in d.iterdir():
            skill = load_skill(entry)  # 디렉토리만 통과
            if skill is None:
                continue
            seen[skill.name] = skill

    return list(seen.values())
```

**디렉토리 형식이 강제인 이유**: `${CLAUDE_SKILL_DIR}`가 유일하게 잘 정의되려면 스킬 = 디렉토리 한 개라는 1:1 대응이 필요하다. 부속 스크립트를 **어디서 찾는지**를 정의하기 위해 디렉토리 경계가 있어야 한다.

### 2단계: SkillTool — 모델이 부르는 도구

9.3의 Tool Protocol을 따르는 도구 한 개로 구현한다. **모델이 도구 목록에서 SkillTool을 발견해서 부르면** — 그 안에서 매칭되는 스킬의 본문을 로드한다.

```python
# src/mini_claude/tools/skill_tool.py
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any
from pydantic import BaseModel, Field
from .base import Tool, ToolContext
from ..skills.skill_md import Skill


# 진짜 코드의 SKILL_BUDGET_CONTEXT_PERCENT = 0.01
SKILL_BUDGET_CHARS = 8_000
MAX_LISTING_DESC_CHARS = 250


class SkillToolInput(BaseModel):
    skill: str = Field(description="The name of the skill to invoke")
    args: str = Field(default="", description="Optional arguments")


def format_skill_listing(skills: list[Skill]) -> str:
    """모델이 보는 스킬 목록.

    1% 컨텍스트 예산 안에서 한 항목당 250자 cap.
    스킬 목록은 *발견용* — 매칭을 결정할 짧은 설명이면 충분하다.
    """
    lines = []
    total_chars = 0

    for skill in skills:
        if skill.disable_model_invocation:
            continue  # 모델 게이트

        # description + when_to_use 합치기
        if skill.when_to_use:
            desc = f"{skill.description} - {skill.when_to_use}"
        else:
            desc = skill.description

        if len(desc) > MAX_LISTING_DESC_CHARS:
            desc = desc[:MAX_LISTING_DESC_CHARS - 1] + "…"

        line = f"- {skill.name}: {desc}"

        if total_chars + len(line) > SKILL_BUDGET_CHARS:
            break
        lines.append(line)
        total_chars += len(line) + 1

    return "\n".join(lines)


@dataclass
class SkillTool:
    """모델이 발견해서 부르는 SkillTool.

    9.3의 Tool Protocol을 따르므로 다른 도구와 같은 풀에 들어간다.
    """

    name: str = "Skill"
    skills: list[Skill] = field(default_factory=list)
    input_model: type[BaseModel] = SkillToolInput

    @property
    def description(self) -> str:
        """모델이 보는 시스템 프롬프트.

        진짜 코드 prompt.ts:173의 핵심 두 가지를 그대로 옮긴다:
        1. slash command = skill 동일시
        2. BLOCKING REQUIREMENT — 말로만 언급하지 말고 반드시 호출
        """
        listing = format_skill_listing(self.skills)
        return f"""Execute a skill within the main conversation.

When users ask you to perform tasks, check if any of the available skills
match. Skills provide specialized capabilities and domain knowledge.

When users reference a "slash command" or "/<something>", they are referring
to a skill. Use this tool to invoke it.

Available skills:
{listing}

Important:
- When a skill matches the user's request, this is a BLOCKING REQUIREMENT:
  invoke the relevant Skill tool BEFORE generating any other response.
- NEVER mention a skill without actually calling this tool.
"""

    def is_read_only(self) -> bool:
        return False

    def is_destructive(self) -> bool:
        return False

    def permission_summary(self, args: dict[str, Any]) -> str:
        return args.get("skill", "")

    async def call(self, args: dict[str, Any], context: ToolContext) -> str:
        skill_name = args["skill"]
        skill_args = args.get("args", "")

        skill = next(
            (s for s in self.skills if s.name == skill_name),
            None,
        )
        if skill is None:
            return f"Skill not found: {skill_name}"

        if skill.disable_model_invocation:
            return f"Skill {skill_name} is not available for model invocation."

        prompt = skill.get_prompt(skill_args)
        return f"Skill {skill_name} loaded. Execute the following task:\n\n{prompt}"
```

**핵심 구조**: SkillTool의 `description` 프로퍼티가 모델 시스템 프롬프트에 스킬 목록을 노출한다. 모델이 매칭을 결정하면 `call()`이 해당 스킬의 본문을 로드해서 돌려준다. 모델은 그 텍스트를 보고 작업을 수행한다.

### 3단계: main.py에 연결

10.1 의 main_async() 위에 _두 자리_ 만 더한다 — _SkillTool 도구 풀 추가_ + _슬래시 분기에서 스킬을 명령의 fallback 으로 찾기_. 10.1 의 시그니처, try/except, parts split, active_permissions 임시 사본, user_input 갈아끼움, query 한 자리 호출 모두 그대로 보존.

```python
# src/mini_claude/main.py — 10.1 위의 add-up
from .skills.skill_md import discover_skills              # ⭐ 추가 import
from .tools.skill_tool import SkillTool                   # ⭐ 추가 import


async def main_async() -> None:
    # ... 9.5 의 argparse / VERTEX_PROJECT_ID 검사 / state / permissions / agent_tool ...
    # ... 10.1 의 md_commands = discover_commands(...) ...

    # ⭐ 추가 ①: 스킬 발견 + SkillTool 도구 풀에 추가 (한 줄)
    skills = discover_skills(cwd=args.cwd, home=Path.home())
    parent_tools.append(SkillTool(skills=skills))

    print("mini-claude 시작 (Ctrl+D로 종료)")
    while True:
        try:
            user_input = await asyncio.to_thread(input, "> ")
        except (EOFError, KeyboardInterrupt):
            print("\nbye.")
            return
        if not user_input.strip():
            continue

        active_permissions = permissions

        # ⭐ 추가 ②: 슬래시면 _10.1 의 마크다운 명령_ 또는 _user_invocable 스킬_ 에서 찾기
        if user_input.startswith("/"):
            parts = user_input[1:].split(maxsplit=1)
            cmd_name = parts[0]
            cmd_args = parts[1] if len(parts) > 1 else ""

            # MarkdownCommand 와 Skill 은 둘 다 get_prompt(args)·allowed_tools 인터페이스
            # → 한 변수 cmd 로 통합 (10.2 의 _slash command = skill_ 동일시의 코드 차원)
            cmd = next((c for c in md_commands if c.name == cmd_name), None)
            if cmd is None:
                cmd = next(
                    (s for s in skills if s.name == cmd_name and s.user_invocable),
                    None,
                )
            if cmd is None:
                print(f"Unknown command: /{cmd_name}")
                continue

            user_input = cmd.get_prompt(cmd_args)  # 본문 → LLM user 메시지
            active_permissions = PermissionEngine(
                allow_rules=permissions.allow_rules | set(cmd.allowed_tools),
                deny_rules=permissions.deny_rules,
            )

        # 9.5 의 query() 호출 — 10.1 그대로. 일반 채팅이면 모델이 SkillTool 을 _알아서_ 부른다
        async for chunk in query(
            user_input=user_input,
            tools=parent_tools,
            permissions=active_permissions,
            cwd=str(args.cwd),
            state=state,
        ):
            if isinstance(chunk, TextDelta):
                print(chunk.text, end="", flush=True)
            elif isinstance(chunk, ToolUseStarted):
                print(f"\n[{chunk.name}] {chunk.input}", flush=True)
            elif isinstance(chunk, TurnDone):
                print()
```

**두 자리만 본다** — 추가 ① 은 _스킬 발견 + 도구 풀 한 줄 합류_, 추가 ② 는 _슬래시 분기에 스킬 fallback 한 단계_. `query()` 호출은 10.1 그대로 한 자리. **`parent_tools.append(SkillTool(...))`** 한 줄이 _모델의 자동 발견 경로_ 를 열고, 슬래시 분기의 `cmd` 통합이 _사용자 invoke 경로_ 에서 _마크다운 명령과 스킬을 같은 인터페이스로_ 다룬다. **9.3 의 Tool Protocol 이 SkillTool 도, MarkdownCommand 와 Skill 의 덕타이핑도 모두 받아준다**.

## 두 invoke 경로의 교차

같은 스킬이 두 가지 방법으로 불린다:

```
경로 1: 사용자가 /pr-review 를 직접 친다
        → user_invocable: true 면 자동완성에 등장
        → get_prompt() → query()

경로 2: 사용자가 "이 PR 봐줘" 라고 자연어로 말한다
        → 모델이 시스템 프롬프트의 스킬 목록에서 매칭
        → SkillTool.call() → get_prompt() → 결과를 모델이 처리
```

**끝은 같다** — 둘 다 같은 `SKILL.md` 본문이 LLM으로 흘러간다. 차이는 누가 부르느냐.

두 게이트 필드가 독립적으로 제어한다:

| `user-invocable` | `disable-model-invocation` | 결과 |
|---|---|---|
| `true` | `false` | 둘 다 부를 수 있음 (기본값) |
| `true` | `true` | 사용자만 — 모델은 발견 못 함 |
| `false` | `false` | 모델만 — 사용자 자동완성에 안 보임 |
| `false` | `true` | 둘 다 못 부름 — 비활성 |

- **사용자만**: `debug`, `batch` 같은 메타 도구. 모델이 자동으로 부르면 오용 가능성.
- **모델만**: 백그라운드 자동화. 사용자 자동완성을 어지럽히지 않게.

## `when_to_use` — 좋은 트리거의 힘

모델이 스킬을 매칭하는 데 보는 것은 **`description` + `when_to_use`**뿐이다.

```yaml
# 나쁜 예
description: 코드 리뷰

# 좋은 예
description: PR의 변경사항을 보안 패턴 + 스타일 룰로 검사한다
when_to_use: 사용자가 "리뷰 좀", "PR 봐줘", "코드 체크" 같은 표현을 쓸 때
```

**`description`은 무엇인지, `when_to_use`는 언제 쓰는지**. 역할이 명확히 분리돼 있다. `when_to_use`에 **사용자가 실제로 쓸 자연어 표현**을 박아두면 매칭 정확도가 올라간다. 모델은 이 두 줄짜리 설명만 보고 결정하니까 — **짧지만 정확한 게 핵심**.

## 1% 컨텍스트 예산

스킬 목록이 시스템 프롬프트에 들어간다. 그런데 스킬이 50개, 100개가 되면?

```python
SKILL_BUDGET_CHARS = 8_000              # 컨텍스트의 ~1% (200k × 4 chars/token × 1%)
MAX_LISTING_DESC_CHARS = 250            # 한 항목 cap
```

**스킬 목록은 발견용이지 내용 전달용이 아니다**. 매칭 정확도는 짧은 description으로 충분하다. 긴 `when_to_use`는 prompt cache의 turn-1 cache_creation 토큰만 낭비한다. 모델이 "이 스킬이 매칭한다"고 결정하면 — 그때 `call()`이 전체 본문을 로드한다.

예산이 부족하면 뒤쪽 스킬부터 잘린다. 진짜 코드에서는 번들 스킬 14개를 항상 보호하고 사용자 스킬부터 자른다.

"/review-pr"), they are referring to a skill. Use this tool to invoke it.
```

**충격 1: 슬래시 명령과 스킬이 모델 입장에서 이미 같은 것**. 사용자가 `/commit`을 친다 → 모델에게는 "commit 스킬을 invoke 하라"로 전달된다. `.claude/commands/`와 `.claude/skills/`는 사람의 멘탈 모델을 위한 구분이지, 모델에게는 같은 것. 10.1의 `commands_DEPRECATED` 폭로의 진짜 의미.

```
this is a BLOCKING REQUIREMENT: invoke the relevant Skill tool BEFORE
generating any other response. NEVER mention a skill without actually
calling this tool.
```

**충격 2: 모델이 말로만 스킬을 언급하는 걸 대문자로 막는다**. "이 작업은 verify 스킬로 하면 좋겠어"같은 말만 하는 건 안 됨. 반드시 도구 호출로 이어져야 함.

이건 **흔한 LLM 프롬프트 엔지니어링 함정** — 모델이 도구를 부르는 대신 도구 이름을 텍스트로 답하는 케이스 — 을 **BLOCKING REQUIREMENT + NEVER**로 막는다. **대문자 강조에 LLM이 민감하다**는 사실을 활용한 production 트릭.

"""staged diff 의 파일 확장자별 변경 통계."""
from __future__ import annotations
import subprocess
from collections import Counter


def main() -> None:
    result = subprocess.run(
        ["git", "diff", "--cached", "--name-only"],
        capture_output=True, text=True, check=True,
    )
    files = [f for f in result.stdout.splitlines() if f]
    by_ext = Counter(
        f.rsplit(".", 1)[-1] if "." in f else "(none)" for f in files
    )
    for ext, count in by_ext.most_common():
        print(f"{ext}: {count} files")


if __name__ == "__main__":
    main()
EOF
```

## 핵심 정리

- **스킬 = 디렉토리 한 개**. `.claude/skills/my-skill/SKILL.md` — 평탄 .md 형식은 명시적으로 거부된다. 디렉토리가 단위인 이유: `${CLAUDE_SKILL_DIR}`가 유일하게 잘 정의되려면 스킬과 디렉토리의 1:1 대응이 필요하다. 부속 스크립트 + 설정이 한 곳에 산다.
- **두 invoke 경로**: 사용자 `/myskill` (slash command)과 모델 SkillTool (자동 발견). **끝은 같다** — 둘 다 같은 `SKILL.md` 본문이 LLM으로 흘러간다. 두 게이트 필드(`user-invocable`, `disable-model-invocation`)가 독립적으로 제어.
- **SkillTool은 9.3의 Tool Protocol을 그대로 따른다** — `name`, `description`, `input_model`, `call`. 다른 도구와 같은 풀에 들어간다. `parent_tools.append(skill_tool)` 한 줄이 전부.
- **`when_to_use`가 자동 발견의 핵심**: `description`은 무엇인지, `when_to_use`는 언제 쓰는지. 사용자가 실제로 쓸 자연어 표현을 박아두면 매칭 정확도가 올라간다.
- **1% 컨텍스트 예산**: 스킬 목록은 발견용. 8000자 안에 한 항목당 250자 cap. 모델이 매칭을 결정하면 그때 전체 본문을 로드한다 — lazy load.
- **진짜 코드의 충격**: 슬래시 명령과 스킬은 모델 입장에서 이미 같은 것. "BLOCKING REQUIREMENT" + "NEVER mention without calling" — 모델이 말로만 언급하는 흔한 LLM 함정을 대문자 강조로 막는다.
- **10.1 회수**: `parse_frontmatter()`, `substitute_arguments()`, git 경계 walker — 전부 재사용. 스킬 시스템은 명령 시스템 위에 **한 겹만 더 쌓은 것**.
